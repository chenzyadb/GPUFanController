cmake_minimum_required(VERSION 3.20)
project(GPUFanController VERSION 1.0 LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "GPUFanController requires Windows and an NVIDIA display driver.")
endif()

enable_language(RC)
configure_file(src/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/app.rc" PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app-icon.ico;${CMAKE_CURRENT_SOURCE_DIR}/src/resource.h")

find_package(OpenGLGUI CONFIG REQUIRED PATHS "${CMAKE_CURRENT_SOURCE_DIR}/gui_lib" NO_DEFAULT_PATH)
add_executable(GPUFanController WIN32 src/main.cpp src/fan_app.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_compile_features(GPUFanController PRIVATE cxx_std_17)
target_compile_definitions(GPUFanController PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(GPUFanController PRIVATE OpenGLGUI::Static shell32 comctl32)
if(MSVC)
    target_compile_options(GPUFanController PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8>)
else()
    target_compile_options(GPUFanController PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
if(MINGW)
    target_link_options(GPUFanController PRIVATE -static)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(controller_tests tests/controller_tests.cpp)
    target_compile_features(controller_tests PRIVATE cxx_std_17)
    target_include_directories(controller_tests PRIVATE src)
    add_test(NAME controller_tests COMMAND controller_tests)

    add_library(nvml_fixture SHARED tests/fake_driver.cpp)
    add_library(nvapi_fixture SHARED tests/fake_driver.cpp)
    foreach(fixture IN ITEMS nvml_fixture nvapi_fixture)
        target_compile_features(${fixture} PRIVATE cxx_std_17)
        target_compile_definitions(${fixture} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        set_target_properties(${fixture} PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fixtures")
        if(MINGW)
            target_link_options(${fixture} PRIVATE -static)
        endif()
    endforeach()
    set_target_properties(nvml_fixture PROPERTIES OUTPUT_NAME nvml)
    set_target_properties(nvapi_fixture PROPERTIES OUTPUT_NAME nvapi64)
    add_executable(nvidia_gpu_tests tests/nvidia_gpu_tests.cpp)
    target_compile_features(nvidia_gpu_tests PRIVATE cxx_std_17)
    target_compile_definitions(nvidia_gpu_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(nvidia_gpu_tests PRIVATE src)
    set_target_properties(nvidia_gpu_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fixtures")
    add_dependencies(nvidia_gpu_tests nvml_fixture nvapi_fixture)
    add_test(NAME nvidia_gpu_tests COMMAND nvidia_gpu_tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME gui_smoke COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/gui_smoke.py"
            $<TARGET_FILE:GPUFanController> $<TARGET_FILE:nvml_fixture> $<TARGET_FILE:nvapi_fixture>)
        set_tests_properties(gui_smoke PROPERTIES TIMEOUT 35 RUN_SERIAL TRUE)
        foreach(scenario IN ITEMS 10 11)
            add_test(NAME startup_denied_${scenario} COMMAND ${Python3_EXECUTABLE}
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/startup_failure_test.py"
                $<TARGET_FILE:GPUFanController> $<TARGET_FILE:nvml_fixture> $<TARGET_FILE:nvapi_fixture> ${scenario})
            set_tests_properties(startup_denied_${scenario} PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE)
        endforeach()
    endif()
endif()
